You are an expert Node.js + front-end engineer. Add a **local HTML visualizer** for my Amap itinerary JSON.

### Goal
Create a small tool that reads `example/output-amap.json` and generates a standalone HTML file `output-map.html` that visualizes the trip on **Amap JS (v2.0)** with default style. The map should:
- Show markers grouped by day
- Draw a polyline connecting places in each day (simple order line)
- Provide a left sidebar to pick Day (filter markers+lines)
- Show an info window (name, address, notes, type, status)
- Work offline except the Amap JS script itself

**IMPORTANT:** Amap JS requires a Web JS key, which is different from the Web Service key. In the final HTML, include:
`<script src="https://webapi.amap.com/maps?v=2.0&key=AMAP_JS_KEY_HERE"></script>`
and instruct the user to replace `AMAP_JS_KEY_HERE` with their real JS key.

### Files to create

1) `scripts/generateAmapHtml.js`
- Node script that:
  - reads JSON from `example/output-amap.json` by default (or allow `--in` path)
  - writes `output-map.html` to project root (or allow `--out`)
  - inlines the JSON as a `<script id="trip-data" type="application/json">...</script>`
  - injects the HTML template (from file 2) and replaces a placeholder `/*__DATA_PLACEHOLDER__*/` with that script tag
- Must be pure ESM and compatible with Node 18+
- CLI usage:
  - `node scripts/generateAmapHtml.js`
  - `node scripts/generateAmapHtml.js --in example/output-amap.json --out output-map.html`

2) `scripts/template.amap.html`
- A minimal, clean HTML that:
  - Includes `<script src="https://webapi.amap.com/maps?v=2.0&key=AMAP_JS_KEY_HERE"></script>`
  - Has a left sidebar for days, main area for the map
  - Style: simple CSS (no frameworks)
  - JS logic:
    - Read JSON from `#trip-data`
    - Normalize place.location (supports "lng,lat" string or {lng,lat} object)
    - Build unique day list: accept either `days` array with `day/title/theme` or any day-like structure; fall back to index if missing
    - For each day: create markers and one polyline (skip null locations)
    - Sidebar lets user switch day; show only that day’s markers + line
    - InfoWindow content shows: name, address, notes, type, status
  - Default center: first valid location found; else China center [104.195, 35.675]
  - Colors: assign a distinct color per day from a small palette

import fs from "fs";
import path from "path";
import { fileURLToPath } from "url";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

function parseArg(flag, def = null) {
  const i = process.argv.indexOf(flag);
  return i > -1 && process.argv[i + 1] ? process.argv[i + 1] : def;
}

const inPath = parseArg("--in", "example/output-amap.json");
const outPath = parseArg("--out", "output-map.html");

const templatePath = path.join(__dirname, "template.amap.html");

if (!fs.existsSync(inPath)) {
  console.error(`❌ Input JSON not found: ${inPath}`);
  process.exit(1);
}
if (!fs.existsSync(templatePath)) {
  console.error(`❌ Template not found: ${templatePath}`);
  process.exit(1);
}

const raw = fs.readFileSync(inPath, "utf-8");
let json;
try {
  json = JSON.parse(raw);
} catch (e) {
  console.error("❌ Invalid JSON in input file:", e.message);
  process.exit(1);
}

const dataScript = `<script id="trip-data" type="application/json">${JSON.stringify(json)}</script>`;
const template = fs.readFileSync(templatePath, "utf-8");
const html = template.replace("/*__DATA_PLACEHOLDER__*/", dataScript);

fs.writeFileSync(outPath, html, "utf-8");
console.log(`✅ Wrote ${outPath}`);
console.log("ℹ️  Open it in your browser, and remember to replace AMAP_JS_KEY_HERE with your real Web JS key inside the HTML.");

    const COLORS = ["#2563eb", "#16a34a", "#ea580c", "#a21caf", "#0d9488", "#b91c1c", "#7c3aed"];

    function readTrip() {
      const el = document.getElementById("trip-data");
      if (!el) return null;
      try { return JSON.parse(el.textContent); } catch { return null; }
    }

    function normLoc(loc) {
      if (!loc) return null;
      if (typeof loc === "string") {
        const [lng, lat] = loc.split(",").map(Number);
        if (Number.isFinite(lng) && Number.isFinite(lat)) return [lng, lat];
        return null;
      }
      if (typeof loc === "object" && loc.lng != null && loc.lat != null) {
        const lng = Number(loc.lng), lat = Number(loc.lat);
        return (Number.isFinite(lng) && Number.isFinite(lat)) ? [lng, lat] : null;
      }
      return null;
    }

    function buildDays(data) {
      // Accepts two schemas:
      // A) { title, days: [ { day?, title?, theme?, notes?, places: [...] } ] }
      // B) { days: [ { theme, notes, places } ] } already structured
      const days = [];
      if (Array.isArray(data.days)) {
        data.days.forEach((d, i) => {
          const title = d.title || d.theme || ("Day " + (d.day || (i+1)));
          days.push({
            id: i,
            title,
            places: Array.isArray(d.places) ? d.places : []
          });
        });
      }
      return days;
    }

    function initMap(center) {
      return new AMap.Map("map", {
        zoom: 11,
        center: center || [104.195, 35.675], // fallback: China center
        viewMode: "2D"
      });
    }

    function fitMapTo(map, lnglats) {
      const bounds = new AMap.Bounds(
        [Math.min(...lnglats.map(p=>p[0])), Math.min(...lnglats.map(p=>p[1]))],
        [Math.max(...lnglats.map(p=>p[0])), Math.max(...lnglats.map(p=>p[1]))]
      );
      map.setBounds(bounds, false, [40, 40, 40, 40]);
    }

    (function main() {
      const data = readTrip();
      if (!data) { alert("未读取到行程数据"); return; }

      const dayList = buildDays(data);

      // initial center = first valid point
      let first = null;
      for (const d of dayList) {
        for (const p of d.places) {
          const ll = normLoc(p.location);
          if (ll) { first = ll; break; }
        }
        if (first) break;
      }

      const map = initMap(first);

      // create sidebar
      const daysEl = document.getElementById("days");
      const legendEl = document.getElementById("legend");
      const groups = [];

      dayList.forEach((d, idx) => {
        const color = COLORS[idx % COLORS.length];
        const btn = document.createElement("button");
        btn.className = "day-btn";
        btn.innerHTML = \`<strong>\${d.title}</strong>\`;
        daysEl.appendChild(btn);

        const markers = [];
        const path = [];
        const info = new AMap.InfoWindow({ offset: new AMap.Pixel(0, -22) });

        d.places.forEach((p, i) => {
          const ll = normLoc(p.location);
          if (!ll) return;
          path.push(ll);
          const marker = new AMap.Marker({
            position: ll,
            label: { content: String(i+1), direction: "top" }
          });
          marker.on("click", () => {
            const html = `
              <div style="min-width:220px">
                <div style="font-weight:600;margin-bottom:4px;">\${p.name || "未命名"}</div>
                <div style="color:#6b7280;font-size:12px;margin-bottom:6px;">\${p.address || ""}</div>
                <div style="font-size:12px;line-height:1.4;">
                  \${p.notes ? ("📝 " + p.notes) : ""}
                  \${p.type ? ("<br/>🏷️ " + p.type) : ""}
                  \${p.status ? ("<br/>📌 " + p.status) : ""}
                </div>
              </div>
            `;
            info.setContent(html);
            info.open(map, marker.getPosition());
          });
          markers.push(marker);
        });

        let polyline = null;
        if (path.length >= 2) {
          polyline = new AMap.Polyline({
            path,
            strokeColor: color,
            strokeWeight: 4,
            strokeOpacity: 0.9
          });
        }

        groups.push({ btn, markers, polyline, color, title: d.title, path });
      });

      function showGroup(i) {
        // clear all
        groups.forEach((g, idx) => {
          g.btn.classList.toggle("active", idx === i);
          g.markers.forEach(m => m.setMap(null));
          if (g.polyline) g.polyline.setMap(null);
        });
        // add selected
        const g = groups[i];
        g.markers.forEach(m => m.setMap(map));
        if (g.polyline) g.polyline.setMap(map);

        const points = g.path;
        if (points.length) fitMapTo(map, points);
      }

      // mount buttons
      groups.forEach((g, idx) => {
        g.btn.onclick = () => showGroup(idx);
      });

      // legend
      legendEl.innerHTML = groups.map((g, idx) =>
        \`<span class="dot" style="background:\${g.color}"></span>\${g.title}\`
      ).join("<br/>");

      // default select first day if any
      if (groups.length) showGroup(0);
    })();
