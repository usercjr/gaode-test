// src/fetchTransit.js
// 多策略公交路线查询（高德 Web 服务 API）
// 文档：/v3/direction/transit/integrated

import { AMAP_WEB_SERVICE_KEY } from "./constants.js";

/** 将 "lng,lat;lng,lat;..." 解析为 [[lng,lat], ...] */
function parsePolyline(str) {
  if (!str) return [];
  return str.split(";").map(s => {
    const [lng, lat] = s.split(",").map(Number);
    return [lng, lat];
  }).filter(([lng, lat]) => Number.isFinite(lng) && Number.isFinite(lat));
}

/** 取一条换乘方案中的整段“可绘制路径”（步行 polyline + 公交 polyline） */
function flattenTransitPath(transit) {
  const segments = transit?.segments || [];
  const path = [];
  for (const seg of segments) {
    // 步行段
    if (seg.walking?.steps) {
      for (const st of seg.walking.steps) path.push(...parsePolyline(st.polyline));
    }
    // 公交/地铁段（可能多条 buslines）
    if (seg.bus?.buslines) {
      for (const line of seg.bus.buslines) path.push(...parsePolyline(line.polyline));
    }
    // 地铁入口/出口的点无需额外处理（已含于 polyline）
  }
  return path;
}

/**
 * 查询两点之间的公交换乘方案（返回前三方案或对应策略方案）
 * @param {[number,number]} origin [lng,lat]
 * @param {[number,number]} dest [lng,lat]
 * @param {Object} opts { city, strategy }
 * @returns {Promise<Array<{strategy:number, label:string, path:number[][], raw:any}>>}
 */
export async function fetchTransitRoutes(origin, dest, { city, strategy }) {
  const key = AMAP_WEB_SERVICE_KEY;
  if (!key || key.includes("YOUR_AMAP_WEB_SERVICE_KEY")) {
    throw new Error("请在 src/constants.js 中填写 AMAP_WEB_SERVICE_KEY");
  }
  const originStr = `${origin[0]},${origin[1]}`;
  const destStr   = `${dest[0]},${dest[1]}`;

  const url = new URL("https://restapi.amap.com/v3/direction/transit/integrated");
  url.searchParams.set("origin", originStr);
  url.searchParams.set("destination", destStr);
  url.searchParams.set("city", city);
  url.searchParams.set("key", key);
  url.searchParams.set("output", "JSON");
  url.searchParams.set("strategy", String(strategy)); // 0/1/2/3/5

  const res = await fetch(url.toString());
  const data = await res.json();

  if (data.status !== "1" || !data.route?.transits?.length) {
    // 可能无公交方案（城市/时间/点位限制），返回空
    return [];
  }

  // 只取若干条高德返回的 transits（通常已按该 strategy 排序）
  // 这里取前 1 条就够（你选择“每个策略切一条方案显示”）
  const t0 = data.route.transits[0];
  const path = flattenTransitPath(t0);

  return [{
    strategy,
    label: "",   // 标签在渲染层根据策略码映射
    path,
    raw: t0
  }];
}

// src/renderRoutes.js
// 根据 Day 的相邻点，按当前策略为每一段生成公交路线，并在地图上绘制

import { fetchTransitRoutes } from "./fetchTransit.js";
import { STRATEGY_COLORS } from "./constants.js";

export class TransitRenderer {
  /**
   * @param {AMap.Map} map
   * @param {Object} opts { city, strategy }
   */
  constructor(map, opts) {
    this.map = map;
    this.city = opts.city;
    this.strategy = opts.strategy; // 0/1/2/3/5
    this.polylines = [];           // 当前已绘制的线路
  }

  setStrategy(strategy) {
    this.strategy = strategy;
  }

  clear() {
    this.polylines.forEach(pl => pl.setMap(null));
    this.polylines = [];
  }

  /** 将相邻点对转换为 [ [lng,lat], [lng,lat] ] */
  static buildAdjacentPairs(places) {
    const norm = p => {
      if (!p?.location) return null;
      if (typeof p.location === "string") {
        const [lng, lat] = p.location.split(",").map(Number);
        return Number.isFinite(lng) && Number.isFinite(lat) ? [lng, lat] : null;
      }
      if (p.location.lng != null && p.location.lat != null) {
        const lng = Number(p.location.lng), lat = Number(p.location.lat);
        return Number.isFinite(lng) && Number.isFinite(lat) ? [lng, lat] : null;
      }
      return null;
    };
    const pts = places.map(norm).filter(Boolean);
    const pairs = [];
    for (let i = 0; i < pts.length - 1; i++) pairs.push([pts[i], pts[i + 1]]);
    return pairs;
  }

  /** 按当前策略为某一天的相邻点绘制公交路线 */
  async renderDay(places) {
    this.clear();
    const pairs = TransitRenderer.buildAdjacentPairs(places);
    if (!pairs.length) return;

    for (const [o, d] of pairs) {
      const routes = await fetchTransitRoutes(o, d, { city: this.city, strategy: this.strategy });
      if (!routes.length) continue;
      const r = routes[0]; // 该策略下的最优方案
      const polyline = new AMap.Polyline({
        path: r.path,
        strokeColor: STRATEGY_COLORS[this.strategy] || "#2563eb",
        strokeWeight: 5,
        strokeOpacity: 0.95,
      });
      polyline.setMap(this.map);
      this.polylines.push(polyline);
    }
  }
}

// src/renderMap.js
// 负责：读取行程数据、初始化地图与 UI、切换 Day、切换公交策略并调用 TransitRenderer

import { DEFAULT_CITY, TRANSIT_STRATEGIES } from "./constants.js";
import { TransitRenderer } from "./renderRoutes.js";

function readTrip() {
  const el = document.getElementById("trip-data");
  if (!el) return null;
  try { return JSON.parse(el.textContent); } catch { return null; }
}

function normLoc(loc) {
  if (!loc) return null;
  if (typeof loc === "string") {
    const [lng, lat] = loc.split(",").map(Number);
    if (Number.isFinite(lng) && Number.isFinite(lat)) return [lng, lat];
    return null;
  }
  if (typeof loc === "object" && loc.lng != null && loc.lat != null) {
    const lng = Number(loc.lng), lat = Number(loc.lat);
    return (Number.isFinite(lng) && Number.isFinite(lat)) ? [lng, lat] : null;
  }
  return null;
}

function buildDays(data) {
  const days = [];
  if (Array.isArray(data.days)) {
    data.days.forEach((d, i) => {
      const title = d.title || d.theme || ("Day " + (d.day || (i + 1)));
      days.push({
        id: i,
        title,
        places: Array.isArray(d.places) ? d.places : []
      });
    });
  } else {
    // 兼容纯数组
    days.push({ id: 0, title: "Day 1", places: data.places || [] });
  }
  return days;
}

function initMap(center) {
  return new AMap.Map("map", {
    zoom: 12,
    center: center || [116.39739, 39.90886], // fallback: 北京天安门
    viewMode: "2D"
  });
}

function fitMapTo(map, lnglats) {
  if (!lnglats.length) return;
  const bounds = new AMap.Bounds(
    [Math.min(...lnglats.map(p => p[0])), Math.min(...lnglats.map(p => p[1]))],
    [Math.max(...lnglats.map(p => p[0])), Math.max(...lnglats.map(p => p[1]))]
  );
  map.setBounds(bounds, false, [40, 40, 40, 40]);
}

function createStrategyButtons(container, onPick) {
  const frag = document.createDocumentFragment();
  TRANSIT_STRATEGIES.forEach((s, idx) => {
    const btn = document.createElement("button");
    btn.className =
      "px-3 py-1.5 text-sm rounded-md border bg-white hover:bg-gray-50 transition " +
      (idx === 0 ? "ring-2 ring-blue-200 border-blue-500" : "border-gray-200");
    btn.textContent = s.label;
    btn.onclick = () => onPick(s.code, btn);
    frag.appendChild(btn);
  });
  container.innerHTML = "";
  container.appendChild(frag);
}

function createDayButtons(container, dayList, onPick) {
  const frag = document.createDocumentFragment();
  dayList.forEach((d, idx) => {
    const btn = document.createElement("button");
    btn.className =
      "w-full text-left px-3 py-2 rounded-lg border bg-white hover:bg-gray-50 transition mb-2 " +
      (idx === 0 ? "ring-2 ring-blue-200 border-blue-500" : "border-gray-200");
    btn.innerHTML = `<div class="font-semibold">${d.title}</div><div class="text-xs text-gray-500">${d.places.length} 个点</div>`;
    btn.onclick = () => onPick(idx, btn);
    frag.appendChild(btn);
  });
  container.innerHTML = "";
  container.appendChild(frag);
}

(function main() {
  const data = readTrip();
  if (!data) { alert("未读取到行程数据"); return; }

  const days = buildDays(data);

  // 首个有效点作为中心
  let first = null;
  for (const d of days) {
    for (const p of d.places) {
      const ll = normLoc(p.location);
      if (ll) { first = ll; break; }
    }
    if (first) break;
  }

  const map = initMap(first);

  // 左侧 Day 列表
  const dayListEl = document.getElementById("days");
  // 右上策略按钮
  const strategyEl = document.getElementById("strategyBtns");

  let currentDayIdx = 0;
  let currentStrategy = TRANSIT_STRATEGIES[0].code;

  // 地点标记（按 Day 切换）
  let liveMarkers = [];
  function clearMarkers() {
    liveMarkers.forEach(m => m.setMap(null));
    liveMarkers = [];
  }
  function renderMarkers(day) {
    clearMarkers();
    const pts = [];
    const info = new AMap.InfoWindow({ offset: new AMap.Pixel(0, -22) });
    day.places.forEach((p, i) => {
      const ll = normLoc(p.location);
      if (!ll) return;
      pts.push(ll);
      const marker = new AMap.Marker({
        position: ll,
        label: { content: String(i + 1), direction: "top" }
      });
      marker.on("click", () => {
        const html = `
          <div style="min-width:220px">
            <div style="font-weight:600;margin-bottom:4px;">${p.name || "未命名"}</div>
            <div style="color:#6b7280;font-size:12px;margin-bottom:6px;">${p.address || ""}</div>
            <div style="font-size:12px;line-height:1.4;">
              ${p.notes ? ("📝 " + p.notes) : ""}
              ${p.type ? ("<br/>🏷️ " + p.type) : ""}
              ${p.status ? ("<br/>📌 " + p.status) : ""}
            </div>
          </div>
        `;
        info.setContent(html);
        info.open(map, marker.getPosition());
      });
      liveMarkers.push(marker);
    });
    liveMarkers.forEach(m => m.setMap(map));
    if (pts.length) fitMapTo(map, pts);
  }

  // 公交路线渲染器
  const transit = new TransitRenderer(map, { city: DEFAULT_CITY, strategy: currentStrategy });

  async function renderDayAndRoutes(idx) {
    currentDayIdx = idx;
    const day = days[idx];
    renderMarkers(day);
    await transit.renderDay(day.places);
  }

  // 装载 UI
  createDayButtons(dayListEl, days, (idx, btn) => {
    // 按钮激活样式切换
    [...dayListEl.children].forEach(b => b.classList.remove("ring-2", "ring-blue-200", "border-blue-500"));
    btn.classList.add("ring-2", "ring-blue-200", "border-blue-500");
    renderDayAndRoutes(idx);
  });

  createStrategyButtons(strategyEl, async (code, btn) => {
    currentStrategy = code;
    transit.setStrategy(code);
    // 切换按钮样式
    [...strategyEl.children].forEach(b => b.classList.remove("ring-2", "ring-blue-200", "border-blue-500"));
    btn.classList.add("ring-2", "ring-blue-200", "border-blue-500");
    // 重新渲染当前 Day 的公交路线
    await transit.renderDay(days[currentDayIdx].places);
  });

  // 默认渲染第一个 Day
  renderDayAndRoutes(0);
})();

（让它把 example/output-amap.json 注入到 public/map.html 占位符，输出 output-map.html）

// scripts/generateAmapHtml.js
import fs from "fs";
import path from "path";
import { fileURLToPath } from "url";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

function arg(flag, def = null) {
  const i = process.argv.indexOf(flag);
  return i > -1 && process.argv[i + 1] ? process.argv[i + 1] : def;
}

const inPath  = arg("--in", "example/output-amap.json");
const outPath = arg("--out", "output-map.html");
const templatePath = path.join(__dirname, "..", "public", "map.html");

if (!fs.existsSync(inPath))   throw new Error(`Input JSON not found: ${inPath}`);
if (!fs.existsSync(templatePath)) throw new Error(`Template not found: ${templatePath}`);

const raw = fs.readFileSync(inPath, "utf-8");
let json;
try { json = JSON.parse(raw); } catch (e) { throw new Error("Invalid JSON: " + e.message); }

const dataScript = `<script id="trip-data" type="application/json">${JSON.stringify(json)}</script>`;
const template = fs.readFileSync(templatePath, "utf-8");
const html = template.replace("/*__DATA_PLACEHOLDER__*/", dataScript);

fs.writeFileSync(outPath, html, "utf-8");
console.log(`✅ Wrote ${outPath}`);
console.log("ℹ️ 打开该文件前，请先用你的 Web JS Key 替换其中的 AMAP_JS_KEY_HERE。");
